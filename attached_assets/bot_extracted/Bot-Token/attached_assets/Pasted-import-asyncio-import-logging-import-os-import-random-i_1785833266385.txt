async def is_admin(msg: Message) -> bool:
    if msg.chat.type == "private":
        return True
    member = await bot.get_chat_member(msg.chat.id, msg.from_user.id)
    return member.status in (ChatMemberStatus.ADMINISTRATOR, ChatMemberStatus.CREATOR)


async def get_user(msg: Message, command: CommandObject = None):
    if msg.reply_to_message:
        return msg.reply_to_message.from_user
    if command and command.args:
        try:
            uid = int(command.args.split()[0])
            member = await bot.get_chat_member(msg.chat.id, uid)
            return member.user
        except:
            return None
    return None


def parse_time(text: str) -> timedelta:
    if not text:
        return timedelta(hours=1)
    t = text.strip().split()[-1].lower()
    t = t.replace("м", "m").replace("ч", "h").replace("д", "d")
    try:
        if t.endswith("m"):
            return timedelta(minutes=int(t[:-1]))
        if t.endswith("h"):
            return timedelta(hours=int(t[:-1]))
        if t.endswith("d"):
            return timedelta(days=int(t[:-1]))
        return timedelta(minutes=int(t))
    except:
        return timedelta(hours=1)

@dp.message(Command("mute"))
async def cmd_mute(msg: Message, command: CommandObject):
    if not await is_admin(msg):
        return await msg.reply("⛔ Только админы")
    user = await get_user(msg, command)
    if not user:
        return await msg.reply("Ответь на сообщение или укажи ID + время\nПримеры: 1м 5м 30м 2ч 1д")
    until = datetime.now() + parse_time(command.args or "")
    try:
        await bot.restrict_chat_member(
            msg.chat.id, user.id,
            permissions=ChatPermissions(can_send_messages=False),
            until_date=until
        )
        await msg.reply(f"🔇 {user.full_name} замучен до {until.strftime('%d.%m.%Y %H:%M')}")
    except Exception as e:
        await msg.reply(f"❌ {e}")

@dp.message(Command("warn"))
async def cmd_warn(msg: Message, command: CommandObject):

    user = await get_user(msg, command)
    if not user:
        return await msg.reply("Ответь на сообщение или укажи ID")
    chat_id, uid = msg.chat.id, user.id
    warnings_db.setdefault(chat_id, {})
    warnings_db[chat_id][uid] = warnings_db[chat_id].get(uid, 0) + 1

    if count >= 3:
        await bot.ban_chat_member(chat_id, uid)
        warnings_db[chat_id][uid] = 0
        await msg.reply(f"🚫 {user.full_name} получил 3 варна и забанен")
    else:
        await msg.reply(f"⚠️ {user.full_name} — варн {count}/3")

@dp.message(Command("purge"))
async def cmd_purge(msg: Message, command: CommandObject):
    if not await is_admin(msg):
        return await msg.reply("⛔ Только админы")
    try:
        count = min(max(int(command.args or 10), 1), 100)
    except:
        return await msg.reply("Использование: /purge 20")
    deleted = 0
    async for m in bot.get_chat_history(msg.chat.id, limit=count + 1):
        try:
            await m.delete()
            deleted += 1
        except:
            pass
    info = await msg.answer(f"🗑 Удалено: {deleted}")
    await asyncio.sleep(3)
    await info.delete()

async def do_checkin(chat_id: int, user_id: int, reply_msg: Message = None):
    today = date.today()
    streaks.setdefault(chat_id, {})
    data = streaks[chat_id].get(user_id, {"count": 0, "last": None})

    if data["last"] == today:
        text = "Ты уже отмечался сегодня 🔥"
        if reply_msg:
            await reply_msg.reply(text)
        return False

    if data["last"] == today - timedelta(days=1):
        data["count"] += 1
    else:
        data["count"] = 1

    data["last"] = today
    streaks[chat_id][user_id] = data

    text = f"✅ Чекин принят! Твой стрик: <b>{data['count']}</b> дней 🔥"
    if reply_msg:
        await reply_msg.reply(text, parse_mode="HTML")
    return True


@dp.message(Command("checkin"))
async def cmd_checkin(msg: Message):
    await do_checkin(msg.chat.id, msg.from_user.id, msg)


@dp.message(Command("streak"))
async def cmd_streak(msg: Message):
    data = streaks.get(msg.chat.id, {}).get(msg.from_user.id, {"count": 0})
    await msg.reply(f"🔥 Твой текущий стрик: {data['count']} дней")


@dp.message(Command("topstreak"))
async def cmd_topstreak(msg: Message):
    chat_streaks = streaks.get(msg.chat.id, {})
    if not chat_streaks:
        return await msg.reply("Пока никто не имеет стрика")
    top = sorted(chat_streaks.items(), key=lambda x: x[1]["count"], reverse=True)[:10]
    text = "🏆 Топ стриков:\n"
    for i, (uid, data) in enumerate(top, 1):
        text += f"{i}. ID {uid} — {data['count']} дней\n"
    await msg.reply(text)


@dp.message(Command("resetstreak"))
async def cmd_resetstreak(msg: Message, command: CommandObject):
    user = await get_user(msg, command) or msg.from_user
    if user.id != msg.from_user.id and not await is_admin(msg):
        return await msg.reply("⛔ Только свой стрик или админ")
    streaks.get(msg.chat.id, {}).pop(user.id, None)
    await msg.reply("Стрик сброшен")


@dp.message_reaction()
async def on_fire_reaction(event: MessageReactionUpdated):
    if not event.new_reaction:
        return
    has_fire = any(getattr(r, "emoji", None) == "🔥" for r in event.new_reaction)
    if not has_fire:
        return
    user = event.user
    if not user or user.is_bot:
        return
    success = await do_checkin(event.chat.id, user.id)
    if success:
        try:
            await bot.send_message(
                event.chat.id,
                f"🔥 {user.first_name} отметил(а)сь! Стрик продолжается.",
                disable_notification=True
            )
        except:
            pass

@dp.message(F.text.startswith("!"))
async def russian_commands_handler(message: Message):
    text = message.text[1:].strip()
    if not text:
        return
    parts = text.split(maxsplit=1)
    cmd = parts[0].lower()
    args = parts[1] if len(parts) > 1 else ""
    args = args.replace("м", "m").replace("ч", "h").replace("д", "d")
    real_cmd = RUSSIAN_ALIASES.get(cmd)
    if not real_cmd:
        return

    class FakeCommand:
        def __init__(self, args):
            self.args = args

    fake_command = FakeCommand(args)

    handlers = {
        "mute": cmd_mute, "unmute": cmd_unmute, "ban": cmd_ban, "unban": cmd_unban,
        "kick": cmd_kick, "warn": cmd_warn, "unwarn": cmd_unwarn, "purge": cmd_purge,
        "ro": cmd_ro, "pin": cmd_pin, "unpin": cmd_unpin, "title": cmd_title,
        "ship": cmd_ship, "love": cmd_love, "friend": cmd_friend, "marry": cmd_marry,
        "divorce": cmd_divorce, "couple": cmd_couple, "hug": cmd_hug, "kiss": cmd_kiss,
        "checkin": cmd_checkin, "streak": cmd_streak, "topstreak": cmd_topstreak,
        "resetstreak": cmd_resetstreak, "fortune": cmd_fortune, "8ball": cmd_8ball,
        "horoscope": cmd_horoscope, "tarot": cmd_tarot, "predict": cmd_predict,
        "destiny": cmd_destiny, "coin": cmd_coin, "dice": cmd_dice, "rps": cmd_rps,
        "random": cmd_random, "choose": cmd_choose, "rate": cmd_rate,
        "truth": cmd_truth, "dare": cmd_dare, "joke": cmd_joke, "meme": cmd_meme,
        "cat": cmd_cat, "dog": cmd_dog, "quote": cmd_quote, "fact": cmd_fact,
        "compliment": cmd_compliment, "roast": cmd_roast, "ascii": cmd_ascii,
        "password": cmd_password, "uuid": cmd_uuid, "help": cmd_help,
    }

    handler = handlers.get(real_cmd)
    if handler:
        try:
            await handler(message, fake_command)
        except TypeError:
            await handler(message)
